using System;

using System.Collections.Generic;

namespace ProjectPA.ArtPipeline.Editor {
public static class SixPartRigImport {
 [Serializable] public class Spec { public float pixels_per_unit; public Bone[] bones; public string[] draw_order; public Skin[] skins; }
 [Serializable] public class Bone { public string name,parent,path; public float[] unity_local_position,sprite_pivot_normalized; }
 [Serializable] public class Skin { public string key,label; }
 [Serializable] public class ClipSet { public Clip[] clips; }
 [Serializable] public class Clip { public string name; public float duration; public bool loop; public Curve[] curves; }
 [Serializable] public class Curve { public string path,property; public float[][] keys; }
 // JsonUtility does not support nested arrays. The portable keys are converted below.
 [Serializable] public class UnityClipSet { public UnityClip[] clips; }
 [Serializable] public class UnityClip { public string name; public float duration; public bool loop; public UnityCurve[] curves; }
 [Serializable] public class UnityCurve { public string path,property; public Sample[] keys; }
 [Serializable] public class Sample { public float time,value; }
 [Serializable] public class Result { public string unity_version; public int sprites,prefabs,clips,pose_samples; public bool shared_controller,identical_local_poses; public string[] captures; }
 const string Target="Assets/SharedCharacters6_ArmSwing";
 static string Arg(string name,string fallback) {var args=Environment.GetCommandLineArgs(); for(int i=0;i<args.Length-1;i++)if(args[i]==name)return args[i+1];return fallback;}
 [MenuItem("Tools/Project PA/Art/Import Six Part Characters")]
 public static void Import() {Run(false);}
 public static void ValidateBatch() {try{Run(true);EditorApplication.Exit(0);}catch(Exception e){Debug.LogException(e);EditorApplication.Exit(1);}}
 static void BuildSurface(GameObject root, Dictionary<string,Transform> transforms, string source, string skin) {
  var folder=Target+"/"+skin+"/";var texturePath=folder+"torso_mesh.png";
  File.Copy(Path.Combine(source,"sprites",skin,"torso_mesh.png"),texturePath);
  AssetDatabase.ImportAsset(texturePath,ImportAssetOptions.ForceSynchronousImport);
  var importer=(TextureImporter)AssetImporter.GetAtPath(texturePath);
  importer.mipmapEnabled=false;importer.alphaIsTransparency=true;importer.npotScale=TextureImporterNPOTScale.None;importer.textureCompression=TextureImporterCompression.Uncompressed;importer.SaveAndReimport();
  var mat=new Material(Shader.Find("Sprites/Default"));mat.mainTexture=AssetDatabase.LoadAssetAtPath<Texture2D>(texturePath);AssetDatabase.CreateAsset(mat,folder+"Torso.mat");
  var bones=new[]{transforms["body"],transforms["arm_L"],transforms["arm_R"],transforms["leg_L"],transforms["leg_R"]};
  var mesh=new Mesh();var vertices=new List<Vector3>();var uv=new List<Vector2>();var weights=new List<BoneWeight>();var triangles=new List<int>();int nx=37,ny=36;
  for(int j=0;j<ny;j++)for(int i=0;i<nx;i++){
   float x=240+8*i,y=448+8*j;
   vertices.Add(new Vector3((x-384)/256,(710-y)/256,0));uv.Add(new Vector2(x/768,1-y/768));
   bool leg=y>620;float distal=leg?Mathf.InverseLerp(620,688,y):Mathf.InverseLerp(499,557,y)*(1-Mathf.InverseLerp(576,619,y));
   float lateral=leg?1:Mathf.InverseLerp(30,74,Mathf.Abs(x-384));
   float w=leg?Mathf.SmoothStep(0,1,distal):0f;
   weights.Add(new BoneWeight{boneIndex0=0,weight0=1-w,boneIndex1=leg?(x<384?3:4):(x<384?1:2),weight1=w});
   if(i+1<nx&&j+1<ny){int t=j*nx+i;triangles.AddRange(new[]{t,t+1,t+nx,t+1,t+nx+1,t+nx});}
  }
  mesh.SetVertices(vertices);mesh.SetUVs(0,uv);mesh.SetTriangles(triangles,0);mesh.boneWeights=weights.ToArray();
  mesh.bindposes=Array.ConvertAll(bones,b=>b.worldToLocalMatrix*root.transform.localToWorldMatrix);mesh.RecalculateBounds();AssetDatabase.CreateAsset(mesh,folder+"Torso.asset");
  var surface=new GameObject("TorsoSurface");surface.transform.SetParent(root.transform,false);var sr=surface.AddComponent<SkinnedMeshRenderer>();sr.sharedMesh=mesh;sr.bones=bones;sr.rootBone=bones[0];sr.sharedMaterial=mat;sr.updateWhenOffscreen=true;sr.sortingOrder=2;
  foreach(var bone in bones)bone.GetComponent<SpriteRenderer>().enabled=bone.name.StartsWith("arm_");
 }
 static void Run(bool validate) {
  string repo=Directory.GetParent(Directory.GetParent(Application.dataPath).FullName).FullName;
  string source=Arg("-artRigSource",Path.Combine(repo,"ArtPipeline/Rigging/SharedCharacters6_20261105"));
  var spec=JsonUtility.FromJson<Spec>(File.ReadAllText(Path.Combine(source,"shared_rig.json")));
  var animations=JsonUtility.FromJson<UnityClipSet>(File.ReadAllText(Path.Combine(source,"shared_animation_unity.json")));
  if(Directory.Exists(Target))throw new InvalidOperationException(Target);
  Directory.CreateDirectory(Target);
  foreach(var skin in spec.skins) {
   Directory.CreateDirectory(Target+"/"+skin.key);
   foreach(var b in spec.bones)File.Copy(Path.Combine(source,"sprites",skin.key,b.name+".png"),Target+"/"+skin.key+"/"+b.name+".png",false);
  }
  AssetDatabase.Refresh(ImportAssetOptions.ForceSynchronousImport);
  foreach(var skin in spec.skins)foreach(var b in spec.bones) {
   string path=Target+"/"+skin.key+"/"+b.name+".png";
   var importer=(TextureImporter)AssetImporter.GetAtPath(path);
   importer.textureType=TextureImporterType.Sprite;importer.spriteImportMode=SpriteImportMode.Single;
   importer.spritePixelsPerUnit=spec.pixels_per_unit;importer.alphaIsTransparency=true;importer.mipmapEnabled=false;
   importer.textureCompression=TextureImporterCompression.Uncompressed;importer.filterMode=FilterMode.Bilinear;
   var settings=new TextureImporterSettings();importer.ReadTextureSettings(settings);
   settings.spriteAlignment=(int)SpriteAlignment.Custom;settings.spritePivot=new Vector2(b.sprite_pivot_normalized[0],b.sprite_pivot_normalized[1]);settings.spriteMeshType=SpriteMeshType.FullRect;
   importer.SetTextureSettings(settings);importer.SaveAndReimport();
  }
  var clips=new Dictionary<string,AnimationClip>();
  foreach(var input in animations.clips) {
   var clip=new AnimationClip{name=input.name,frameRate=40,wrapMode=WrapMode.Loop};
   foreach(var c in input.curves) {
    var keys=new Keyframe[c.keys.Length];
    for(int i=0;i<keys.Length;i++)keys[i]=new Keyframe(c.keys[i].time,c.keys[i].value);
    var curve=new AnimationCurve(keys);
    for(int i=0;i<keys.Length;i++){AnimationUtility.SetKeyLeftTangentMode(curve,i,AnimationUtility.TangentMode.Linear);AnimationUtility.SetKeyRightTangentMode(curve,i,AnimationUtility.TangentMode.Linear);}
    AnimationUtility.SetEditorCurve(clip,EditorCurveBinding.FloatCurve(c.path,typeof(Transform),c.property),curve);
   }
   var settings=AnimationUtility.GetAnimationClipSettings(clip);settings.loopTime=input.loop;AnimationUtility.SetAnimationClipSettings(clip,settings);
   AssetDatabase.CreateAsset(clip,Target+"/"+input.name+".anim");clips.Add(input.name,clip);
  }
  var controller=AnimatorController.CreateAnimatorControllerAtPath(Target+"/SharedCharacters.controller");controller.AddParameter("Speed",AnimatorControllerParameterType.Float);
  var sm=controller.layers[0].stateMachine;var idle=sm.AddState("Idle");idle.motion=clips["Idle"];var walk=sm.AddState("Walk");walk.motion=clips["Walk"];sm.defaultState=idle;
  var toWalk=idle.AddTransition(walk);toWalk.hasExitTime=false;toWalk.duration=.1f;toWalk.AddCondition(AnimatorConditionMode.Greater,.02f,"Speed");
  var toIdle=walk.AddTransition(idle);toIdle.hasExitTime=false;toIdle.duration=.1f;toIdle.AddCondition(AnimatorConditionMode.Less,.01f,"Speed");
  foreach(var skin in spec.skins) {
   var root=new GameObject("CharacterRig");
   try {
    var transforms=new Dictionary<string,Transform>{{"root",root.transform}};
    foreach(var b in spec.bones) {
     var t=new GameObject(b.name).transform;t.SetParent(transforms[b.parent],false);var xyz=b.unity_local_position;t.localPosition=new Vector3(xyz[0],xyz[1],xyz[2]);transforms.Add(b.name,t);
     var sr=t.gameObject.AddComponent<SpriteRenderer>();sr.sprite=AssetDatabase.LoadAssetAtPath<Sprite>(Target+"/"+skin.key+"/"+b.name+".png");sr.sortingOrder=Array.IndexOf(spec.draw_order,b.name);
     if(sr.sprite==null)throw new InvalidOperationException(b.name);
    }
    BuildSurface(root,transforms,source,skin.key);
    var animator=root.AddComponent<Animator>();animator.runtimeAnimatorController=controller;animator.applyRootMotion=false;
    PrefabUtility.SaveAsPrefabAsset(root,Target+"/"+skin.key+"/CharacterRig.prefab");
   } finally {UnityEngine.Object.DestroyImmediate(root);}
  }
  AssetDatabase.SaveAssets();
  if(!validate)return;
  var instances=new List<GameObject>();
  try {
   foreach(var skin in spec.skins) {
    var prefab=AssetDatabase.LoadAssetAtPath<GameObject>(Target+"/"+skin.key+"/CharacterRig.prefab");var go=(GameObject)PrefabUtility.InstantiatePrefab(prefab);instances.Add(go);
    if(go.GetComponentsInChildren<SpriteRenderer>().Length!=6)throw new InvalidOperationException(skin.key);
    if(go.GetComponent<Animator>().runtimeAnimatorController!=controller)throw new InvalidOperationException("ControllerReference");
   }
   int samples=0;
   foreach(var input in animations.clips)for(int i=0;i<=32;i++) {
    float t=input.duration*i/32;foreach(var go in instances)clips[input.name].SampleAnimation(go,t);
    foreach(var b in spec.bones) {
     var expected=instances[0].transform.Find(b.path);
     if(Mathf.Abs(expected.localPosition.x-b.unity_local_position[0])>1e-5f)throw new InvalidOperationException("RestX_"+b.path);
     var yc=Array.Find(input.curves,c=>c.path==b.path&&c.property=="m_LocalPosition.y");
     var rc=Array.Find(input.curves,c=>c.path==b.path&&c.property=="localEulerAnglesRaw.z");
     if(Mathf.Abs(expected.localPosition.y-yc.keys[i].value)>1e-5f||Mathf.Abs(Mathf.DeltaAngle(expected.localEulerAngles.z,rc.keys[i].value))>.001f)throw new InvalidOperationException("AuthoredCurve_"+b.path);
     for(int k=1;k<instances.Count;k++){var actual=instances[k].transform.Find(b.path);if(Vector3.Distance(expected.localPosition,actual.localPosition)>1e-5f||Quaternion.Angle(expected.localRotation,actual.localRotation)>1e-3f)throw new InvalidOperationException(b.path);}
    }
    samples+=instances.Count;
   }
   clips["Walk"].SampleAnimation(instances[0],0);float z0=instances[0].transform.Find("body/arm_L").localEulerAngles.z;
   clips["Walk"].SampleAnimation(instances[0],.2f);float z1=instances[0].transform.Find("body/arm_L").localEulerAngles.z;if(Mathf.Abs(Mathf.DeltaAngle(z0,z1))<5)throw new InvalidOperationException("WalkBinding");
   var captures=new List<string>();
   var camObject=new GameObject("ArtValidationCamera");var camera=camObject.AddComponent<Camera>();camera.orthographic=true;camera.orthographicSize=1.9f;camera.transform.position=new Vector3(0,1.4f,-10);camera.clearFlags=CameraClearFlags.SolidColor;camera.backgroundColor=new Color(249/255f,246/255f,232/255f);camera.nearClipPlane=.1f;camera.farClipPlane=30;
   for(int i=0;i<instances.Count;i++)instances[i].transform.position=new Vector3((i-1)*3.25f,0,0);
   var rt=new RenderTexture(1152,430,24,RenderTextureFormat.ARGB32);rt.Create();camera.targetTexture=rt;
   foreach(string mode in new[]{"Idle","Walk"})for(int frame=0;frame<8;frame++) {
    float t=(mode=="Idle"?1.6f:.8f)*frame/8;foreach(var go in instances)clips[mode].SampleAnimation(go,t);foreach(var actor in instances){var skin=actor.GetComponentInChildren<SkinnedMeshRenderer>();var baked=actor.transform.Find("CaptureSurface");if(!baked){baked=new GameObject("CaptureSurface").transform;baked.SetParent(actor.transform,false);}var mf=baked.GetComponent<MeshFilter>();if(mf==null){mf=baked.gameObject.AddComponent<MeshFilter>();var rr=baked.gameObject.AddComponent<MeshRenderer>();rr.sharedMaterial=skin.sharedMaterial;rr.sortingOrder=skin.sortingOrder;mf.sharedMesh=new Mesh();skin.enabled=false;}skin.BakeMesh(mf.sharedMesh);}camera.Render();RenderTexture.active=rt;var tex=new Texture2D(1152,430,TextureFormat.RGB24,false);tex.ReadPixels(new Rect(0,0,1152,430),0,0);tex.Apply();string name="unity_"+mode.ToLower()+"_"+frame.ToString("D2")+".png";File.WriteAllBytes(Path.Combine(source,"preview",name),tex.EncodeToPNG());captures.Add(name);UnityEngine.Object.DestroyImmediate(tex);
   }
   RenderTexture.active=null;camera.targetTexture=null;rt.Release();UnityEngine.Object.DestroyImmediate(rt);UnityEngine.Object.DestroyImmediate(camObject);
   var result=new Result{unity_version=Application.unityVersion,sprites=18,prefabs=3,clips=clips.Count,pose_samples=samples,shared_controller=true,identical_local_poses=true,captures=captures.ToArray()};File.WriteAllText(Path.Combine(source,"unity_validation.json"),JsonUtility.ToJson(result,true));
   AssetDatabase.ExportPackage(Target,Path.Combine(source,"SharedCharacters6.unitypackage"),ExportPackageOptions.Recurse);
  }finally{foreach(var go in instances)UnityEngine.Object.DestroyImmediate(go);}
 }
}
